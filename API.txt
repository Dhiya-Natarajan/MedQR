const express = require("express"); // -> This returns a function
                                    //  require imports Express Frame Work

app = express() // Now we are calling express, which creates an express application instance (creates the server)

// app is the web server listening to client request
      // here we define the routes

// Synatx : app.Method("PATH", handler) =>{
  //Read from request
  // Perform some DB logic
  // Send the response to the client
}

TO CREATE A PATIENT

app.post("/patient",(req,res)=>{
  const {name, dob} = req.body;

  res.status(201).json({
    message : "patient created",
    name,
    dob
  });

});


TO GET PATIENT

app.get("/patient",(req,res)=>{
  res.json({
    message : "Fetching all patient"
  });
});
